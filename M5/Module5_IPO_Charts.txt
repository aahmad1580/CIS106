M5P1 - Item Total
Input: Quantity
Processing: If quantity >= 1000, unit_price = 3.00; otherwise unit_price = 5.00.
            extended_price = quantity * unit_price
            tax = extended_price * 0.07
            total = extended_price + tax
Output: Quantity, unit price, extended price, tax, total

M5P2 - Item Price
Input: Item, quantity
Processing: If item == "A", unit_price = 10.00; otherwise unit_price = 20.00.
            extended_price = quantity * unit_price
Output: Item, unit price, extended price

M5P3 - Book Shipping
Input: Number of books, cost per book
Processing: order_total = books * cost_per_book
            If order_total > 50.00, shipping = 0.00; otherwise shipping = 25.00.
Output: Order total, shipping charge

M5P4 - Appliance Warranty
Input: Appliance name, cost
Processing: If cost > 1000.00, warranty = cost * 0.10;
            otherwise warranty = cost * 0.05.
            total = cost + warranty
Output: Appliance name, appliance cost, warranty cost, total

M5P5 - Income Tax
Input: Last name, number of dependents, gross income
Processing: adjusted_income = gross_income - dependents * 12000
            If adjusted_income > 50000, tax_rate = 0.20; otherwise tax_rate = 0.10.
            income_tax = adjusted_income * tax_rate
            If income_tax < 0, set income_tax = 100.00.
Output: Last name, gross income, dependents, adjusted gross income, income tax
